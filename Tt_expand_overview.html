Model comparison figures

This notebook builds no new models. It reads the coefficient tables saved by Tt_expand_analysis.ipynb and draws combined figures that compare:

Model Source section Sample Fixed effects (+ (1 | Participant))
Dry only, no Weather §6f Dry sessions only [Pre +] HMI × Direction
Dry + Wet, + Weather §6g Dry + Wet [Pre +] HMI × Direction + Weather
Dry + Wet, HMI × Direction × Weather §6h Dry + Wet [Pre +] HMI × Direction × Weather

All three omit Period and Sequence. The figures are grouped by attention sample (all participants vs. high attention, score ≥ 3) and by Weather model, for all five outcomes. Estimates use the figure_* columns: mixed-model REML values, or, where the mixed-model inference is unusable (L2 trust Post | Pre), the labelled OLS with participant-clustered SEs.

Note: the no-Weather model uses the Dry-only sample, while the Weather models use Dry + Wet. Differences between them therefore reflect both the added Weather terms and the larger sample.

1. Assemble one tidy coefficient table with Polars

Code 
frames = []
for f in FRAMINGS:
    for mkey, (mlabel, pattern, sample_map) in MODELS.items():
        for skey, sfile in sample_map.items():
            d = pl.read_csv(TABLE_DIR / pattern.format(f=f, s=sfile))
            frames.append(d.select(
                "outcome", "term", "figure_estimate", "figure_ci_low", "figure_ci_high", "figure_p",
                "figure_inference_source",
            ).with_columns(
                pl.lit(f).alias("framing"), pl.lit(FRAMINGS[f]).alias("framing_label"),
                pl.lit(mlabel).alias("model"), pl.lit(SAMPLES[skey]).alias("sample"),
                pl.lit(sfile).alias("source_sample"),
            ))

coefs = (
    pl.concat(frames)
    .filter(pl.col("term") != "Intercept")
    .with_columns(pl.when(pl.col("term").str.starts_with("pre_")).then(pl.lit("pre"))
                  .otherwise(pl.col("term")).alias("term_key"))
    .with_columns(
        pl.col("term_key").replace_strict(TERMS).alias("term_label"),
        pl.col("outcome").str.replace(r"^(post|chg)_", "").replace_strict(OUTCOMES).alias("outcome_label"),
        (pl.col("figure_p") < 0.05).alias("significant"),
        pl.col("figure_p").map_elements(lambda p: "< .001" if p < 0.001 else f"{p:.3f}".lstrip("0"),
                                        return_dtype=pl.String).alias("p_text"),
    )
)
assert coefs["figure_estimate"].is_finite().all() and coefs["figure_p"].is_not_null().all()

2. Plot helper

Each panel shows one outcome × one attention sample. Points are estimates with 95% CIs, coloured by Weather model and dodged vertically within each term. Filled points have p < .05; hollow points have p ≥ .05. The dashed line marks zero. Terms that a model does not contain are simply absent for that colour.

Code 
def panel(data, outcome_label, sample, show_legend, show_y, term_order=None):
    d = data.filter((pl.col("outcome_label") == outcome_label) & (pl.col("sample") == sample))
    if term_order is None:
        term_order = [t for t in TERMS.values() if t in d["term_label"].to_list()]
    y = alt.Y("term_label:N", sort=term_order, scale=alt.Scale(domain=term_order), title=None,
              axis=alt.Axis(labels=show_y, ticks=show_y, labelLimit=200))
    color = alt.Color("model:N", scale=alt.Scale(domain=MODEL_ORDER, range=COLORS),
                      legend=alt.Legend(title="Model", orient="bottom", direction="vertical", labelLimit=400) if show_legend else None)
    yoff = alt.YOffset("model:N", sort=MODEL_ORDER)
    base = alt.Chart(d.to_pandas()).encode(y=y, yOffset=yoff, color=color)
    ci = base.mark_rule(strokeWidth=1.6).encode(x=alt.X("figure_ci_low:Q", title="Estimate (95% CI)"),
                                                 x2="figure_ci_high:Q")
    pts = base.mark_point(size=55, strokeWidth=1.6).encode(
        x="figure_estimate:Q",
        fill=alt.condition("datum.significant", color, alt.value("white")),
        tooltip=["model", "sample", "outcome_label", "term_label",
                 alt.Tooltip("figure_estimate:Q", format=".2f", title="b"),
                 alt.Tooltip("figure_ci_low:Q", format=".2f", title="CI low"),
                 alt.Tooltip("figure_ci_high:Q", format=".2f", title="CI high"),
                 alt.Tooltip("p_text:N", title="p"), "figure_inference_source", "source_sample"])
    zero = alt.Chart().mark_rule(strokeDash=[4, 3], color="gray").encode(x=alt.datum(0))
    return alt.layer(zero, ci, pts).properties(width=260, height=alt.Step(9), title=f"{outcome_label} — {sample}")


def comparison_chart(framing, outcomes=OUTCOMES.values()):
    data = coefs.filter(pl.col("framing") == framing)
    rows = []
    outcomes = list(outcomes)
    for i, o in enumerate(outcomes):
        rows.append(alt.hconcat(*[panel(data, o, s, show_legend=(i == len(outcomes) - 1 and j == 0), show_y=(j == 0))
                                  for j, s in enumerate(SAMPLES.values())]).resolve_scale(x="shared"))
    return alt.vconcat(*rows).properties(
        title=alt.Title(f"Weather vs. no-weather models — {FRAMINGS[framing]}",
                        subtitle=["Columns: attention sample. Colour: Weather model. Filled = p < .05.",
                                  "No-Weather model: Dry only; Weather models: Dry + Wet. Period and Sequence omitted; (1 | Participant)."])
    ).configure_axis(labelFontSize=10, titleFontSize=11).configure_title(fontSize=13, anchor="start")


def save(chart, stem):
    chart.save(OUT_DIR / f"{stem}.html")
    chart.save(OUT_DIR / f"{stem}.png", scale_factor=2)
    print("saved", OUT_DIR / f"{stem}.png")

5. One figure per outcome (both framings side by side)

These are the same estimates arranged per outcome: columns are framing × attention sample, so one figure shows everything for a single dependent variable.

Code 
def outcome_chart(outcome_label):
    cols = []
    for k, (f, flabel) in enumerate(FRAMINGS.items()):
        data = coefs.filter(pl.col("framing") == f)
        for j, s in enumerate(SAMPLES.values()):
            p = panel(data, outcome_label, s, show_legend=(k == 0 and j == 0), show_y=(k == 0 and j == 0),
                      term_order=list(TERMS.values()))
            cols.append(p.properties(title=[flabel, s], width=200))
    return alt.hconcat(*cols).resolve_scale(x="independent").properties(
        title=alt.Title(f"{outcome_label}: Weather vs. no-weather models",
                        subtitle="No-Weather model: Dry only; Weather models: Dry + Wet. Filled = p < .05.")
    ).configure_axis(labelFontSize=10).configure_title(fontSize=13, anchor="start")

for key, label in OUTCOMES.items():
    ch = outcome_chart(label)
    save(ch, f"weather_comparison_{key}")
    display(ch)
